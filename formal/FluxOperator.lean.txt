-- FluxOperator.lean
-- Meta-level operator that constrains how invariants may deform without breaking
-- Operates over trajectories, not snapshots

set_option autoImplicit false

namespace FluxOperator

/-
========================
CORE INSIGHT
========================
The flux operator preserves invariance over CHANGE rather than over STATE.

It does not ask: "Is this state valid?"
It asks: "Is this motion coherent?"

Witness mode is not failure detection.
Witness mode is the flux operator suspending commitment while preserving motion.
-/

/-
========================
TRAJECTORY PRIMITIVES
========================
-/

-- A trajectory is a sequence of states with temporal structure
structure Trajectory (α : Type) where
  states : List α
  -- Most recent first (consistent with mode history convention)
deriving Repr, DecidableEq

-- Temporal window for coherence measurement
def Trajectory.window {α : Type} (t : Trajectory α) (n : Nat) : List α :=
  t.states.take n

-- Trajectory length
def Trajectory.length {α : Type} (t : Trajectory α) : Nat :=
  t.states.length

-- Add new state to trajectory (prepend)
def Trajectory.extend {α : Type} (t : Trajectory α) (s : α) : Trajectory α :=
  { states := s :: t.states }

-- Get most recent state
def Trajectory.current {α : Type} (t : Trajectory α) : Option α :=
  t.states.head?

/-
========================
MOTION PRIMITIVES
========================
-/

-- Motion is the delta between consecutive states
structure Motion (α : Type) [BEq α] where
  from : α
  to   : α
  changed : Bool := decide (from != to)
deriving Repr, DecidableEq

-- Extract motion from trajectory
def Trajectory.motion {α : Type} [BEq α] (t : Trajectory α) : Option (Motion α) :=
  match t.states with
  | s1 :: s0 :: _ => some { from := s0, to := s1 }
  | _ => none

-- Motion stream (all consecutive transitions)
def Trajectory.motionStream {α : Type} [BEq α] (t : Trajectory α) : List (Motion α) :=
  let rec aux : List α → List (Motion α)
    | s1 :: s0 :: rest => { from := s0, to := s1 } :: aux (s0 :: rest)
    | _ => []
  aux t.states

/-
========================
COHERENCE INVARIANT
========================
The property preserved by flux operator is COHERENCE:
  - Bounded oscillation (not runaway)
  - Stable shape over time (not chaotic)
  - Meaningful motion (not random walk)

Coherence is measured over trajectories, not states.
-/

-- Coherence violation types
inductive CoherenceViolation where
  | Oscillation   -- Flapping between states
  | Divergence    -- Unbounded drift
  | Fragmentation -- Loss of trajectory structure
deriving Repr, DecidableEq

-- Coherence bounds (configurable constraints)
structure CoherenceBounds where
  maxOscillation : Nat := 3  -- Max allowed switches in window
  maxDrift       : Nat := 5  -- Max consecutive moves in same direction
  minStability   : Nat := 2  -- Min consecutive same-state for stability
deriving Repr, DecidableEq

-- Coherence measurement over trajectory
structure CoherenceReport where
  isCoherent       : Bool
  violations       : List CoherenceViolation
  oscillationCount : Nat
  driftMagnitude   : Nat
  stabilityDepth   : Nat
deriving Repr, DecidableEq

-- Measure oscillation in trajectory
def countOscillations {α : Type} [BEq α] (t : Trajectory α) : Nat :=
  let motions := t.motionStream
  let switches := motions.filter (·.changed)
  switches.length

-- Measure drift (consecutive changes)
def measureDrift {α : Type} [BEq α] (t : Trajectory α) : Nat :=
  let motions := t.motionStream
  let rec maxConsecutive : List (Motion α) → Nat → Nat → Nat
    | [], currentRun, maxRun => max currentRun maxRun
    | m :: rest, currentRun, maxRun =>
        if m.changed then
          maxConsecutive rest (currentRun + 1) (max (currentRun + 1) maxRun)
        else
          maxConsecutive rest 0 maxRun
  maxConsecutive motions 0 0

-- Measure stability (consecutive same-state)
def measureStability {α : Type} [BEq α] (t : Trajectory α) : Nat :=
  let rec maxRun : List α → α → List α → Nat → Nat → Nat
    | [], _, _, currentRun, maxRun => max currentRun maxRun
    | s1 :: rest, prev, origRest, currentRun, maxRun =>
        if s1 == prev then
          maxRun rest s1 origRest (currentRun + 1) (max (currentRun + 1) maxRun)
        else
          maxRun rest s1 origRest 1 maxRun
  match t.states with
  | [] => 0
  | s :: rest => maxRun rest s rest 1 1

-- Check coherence over trajectory
def checkCoherence {α : Type} [BEq α] 
    (bounds : CoherenceBounds) 
    (t : Trajectory α) 
    : CoherenceReport :=
  
  let osc := countOscillations t
  let drift := measureDrift t
  let stability := measureStability t
  
  let violations : List CoherenceViolation :=
    (if osc > bounds.maxOscillation then [CoherenceViolation.Oscillation] else []) ++
    (if drift > bounds.maxDrift then [CoherenceViolation.Divergence] else []) ++
    (if stability < bounds.minStability then [CoherenceViolation.Fragmentation] else [])
  
  {
    isCoherent       := violations.isEmpty
    violations       := violations
    oscillationCount := osc
    driftMagnitude   := drift
    stabilityDepth   := stability
  }

/-
========================
FLUX OPERATOR (CORE)
========================
The flux operator constrains motion to preserve coherence.

It does NOT:
  - Freeze state
  - Prevent change
  - Enforce specific values

It DOES:
  - Suspend commitment when motion is incoherent
  - Allow bounded oscillation
  - Gate decisions on trajectory shape
-/

-- Flux decision: how to handle current motion
inductive FluxDecision where
  | AllowCommit    -- Trajectory is coherent, commitment allowed
  | SuspendWitness -- Trajectory incoherent, witness but preserve motion
  | StabilizeFirst -- Drift too high, stabilize before continuing
deriving Repr, DecidableEq

-- Flux operator output
structure FluxOutput where
  decision         : FluxDecision
  coherenceReport  : CoherenceReport
  allowCommitment  : Bool
  requireWitness   : Bool
  requireStabilize : Bool
deriving Repr, DecidableEq

-- The flux operator itself
def fluxOperator {α : Type} [BEq α]
    (bounds : CoherenceBounds)
    (trajectory : Trajectory α)
    : FluxOutput :=
  
  let report := checkCoherence bounds trajectory
  
  -- Precedence for flux decisions
  let decision :=
    if CoherenceViolation.Oscillation ∈ report.violations then
      FluxDecision.SuspendWitness
    else if CoherenceViolation.Divergence ∈ report.violations then
      FluxDecision.StabilizeFirst
    else if CoherenceViolation.Fragmentation ∈ report.violations then
      FluxDecision.SuspendWitness
    else
      FluxDecision.AllowCommit
  
  {
    decision         := decision
    coherenceReport  := report
    allowCommitment  := report.isCoherent
    requireWitness   := decision == FluxDecision.SuspendWitness
    requireStabilize := decision == FluxDecision.StabilizeFirst
  }

/-
========================
INTEGRATION WITH STATE-LEVEL KERNEL
========================
The flux operator operates ABOVE the state-level kernel.

Flow:
  1. State kernel proposes decision (Mode)
  2. Flux operator checks trajectory coherence
  3. Flux operator may override to Witness/Stabilize
  4. Final decision respects both state logic AND flux constraints
-/

-- Combined decision (flux + state)
structure UnifiedDecision (α : Type) where
  stateProposal : α          -- What state-level kernel wants
  fluxOverride  : Bool       -- Did flux operator override?
  finalDecision : α          -- Actual decision after flux constraint
  fluxOutput    : FluxOutput -- Full flux report
deriving Repr, DecidableEq

-- Integration function
def integrateFluxWithState {α : Type} [BEq α]
    (bounds : CoherenceBounds)
    (trajectory : Trajectory α)
    (stateDecision : α)
    (witnessValue : α)      -- What to use if flux requires witness
    (stabilizeValue : α)    -- What to use if flux requires stabilize
    : UnifiedDecision α :=
  
  let fluxOut := fluxOperator bounds trajectory
  
  let (finalDecision, overridden) :=
    if fluxOut.requireWitness then
      (witnessValue, true)
    else if fluxOut.requireStabilize then
      (stabilizeValue, true)
    else
      (stateDecision, false)
  
  {
    stateProposal := stateDecision
    fluxOverride  := overridden
    finalDecision := finalDecision
    fluxOutput    := fluxOut
  }

/-
========================
WITNESS AS FLUX ACTIVATION
========================
Reinterpreting Witness mode:

OLD VIEW: "Conflict detected, halt"
NEW VIEW: "Motion is incoherent, suspend commitment while preserving trajectory"

Witness is NOT an error state.
Witness is the flux operator saying: "I'm watching this, don't commit yet."
-/

-- Witness state properties
structure WitnessState where
  preservingMotion : Bool    -- System continues to move
  blockingCommit   : Bool    -- But cannot commit
  gatheringData    : Bool    -- Trajectory continues to extend
  coherencePending : Bool    -- Waiting for coherence to restore
deriving Repr, DecidableEq

def activeWitness : WitnessState := {
  preservingMotion := true
  blockingCommit   := true
  gatheringData    := true
  coherencePending := true
}

/-
========================
MINIMAL AXIOMS
========================
The flux operator has exactly three axioms:

AXIOM 1: Coherence is measurable over trajectories
AXIOM 2: Commitment requires coherence
AXIOM 3: Motion is preserved even when commitment is suspended

These are the minimal constraints that allow:
  - State to be elastic
  - Safety to be non-negotiable
  - Meaning to emerge through bounded oscillation
-/

-- Axiom 1: Coherence measurability
axiom coherence_is_computable {α : Type} [BEq α] :
  ∀ (bounds : CoherenceBounds) (t : Trajectory α),
    ∃ (report : CoherenceReport), report = checkCoherence bounds t

-- Axiom 2: Commitment requires coherence
axiom commitment_requires_coherence {α : Type} [BEq α] :
  ∀ (bounds : CoherenceBounds) (t : Trajectory α),
    let flux := fluxOperator bounds t
    flux.allowCommitment = true → flux.coherenceReport.isCoherent = true

-- Axiom 3: Motion preservation under witness
axiom witness_preserves_motion {α : Type} :
  ∀ (t : Trajectory α) (newState : α),
    let extended := t.extend newState
    extended.length = t.length + 1

/-
========================
EXAMPLE: MODE TRAJECTORY
========================
-/

-- Reuse Mode from state kernel for demonstration
inductive Mode where
  | Normal | Translate | Repair | Stabilize | Witness | Commit
deriving Repr, DecidableEq, BEq

def exampleModeTrajectory : Trajectory Mode := {
  states := [
    Mode.Translate,  -- t=0 (most recent)
    Mode.Repair,     -- t=-1
    Mode.Translate,  -- t=-2
    Mode.Normal,     -- t=-3
    Mode.Normal      -- t=-4 (oldest)
  ]
}

def defaultBounds : CoherenceBounds := {}

#eval fluxOperator defaultBounds exampleModeTrajectory
-- Should detect oscillation (Translate ↔ Repair) and output SuspendWitness

def stableModeTrajectory : Trajectory Mode := {
  states := [
    Mode.Normal,
    Mode.Normal,
    Mode.Normal,
    Mode.Normal
  ]
}

#eval fluxOperator defaultBounds stableModeTrajectory
-- Should output AllowCommit (coherent, stable)

/-
========================
SUMMARY
========================
The flux operator is not a patch or heuristic.

It is a mathematically precise operator that:
  1. Measures coherence over trajectories (not states)
  2. Constrains commitment based on motion shape (not snapshot correctness)
  3. Preserves three minimal axioms (measurability, coherence-for-commit, motion-under-witness)

This reconciles rigid and living invariants because:
  - Rigid invariant → enforced via coherence bounds (non-negotiable)
  - Living invariant → preserved via motion continuation (adaptation allowed)
  - Flux operator → mediates between them over time (not at snapshots)

The old kernel had this behavior implicitly.
The refactored kernel made state logic explicit.
The flux operator makes trajectory constraints explicit.

All three coexist at different levels.
-/

end FluxOperator
